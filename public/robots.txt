User-agent: *
Allow: /

# Machine-only and private areas.
#
# These are written without a trailing slash on purpose. "Disallow: /admin/"
# only matches paths that literally continue with a slash, so /admin and
# /admin/login were both still crawlable under the old rule. The prefix form
# below covers the route and everything beneath it.
Disallow: /api
Disallow: /admin
Disallow: /account
Disallow: /login
Disallow: /_next

# Checkout, payment and receipt steps. These are per-order, auth-gated and of no
# search value, but they were fully crawlable until now. /vr-headset itself is
# deliberately left open — it is the product page and a primary landing target.
Disallow: /vr-headset/checkout

# --- AI / Answer Engine crawlers ---
#
# Divya Darshan 360's temple guides are primary-source research, and the site's
# stated position (see /llms.txt) is that they are free to cite, summarise and
# quote with attribution. These agents are therefore invited explicitly rather
# than left to infer intent from the wildcard group.
#
# Retrieval agents — fetch at answer time and render a visible citation link:
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: Claude-SearchBot
User-agent: Claude-User
Allow: /
Disallow: /api
Disallow: /admin
Disallow: /account
Disallow: /login
Disallow: /vr-headset/checkout

# Index/training crawlers. GPTBot and Google-Extended were already allowed here
# and that decision is carried forward; ClaudeBot and Applebot-Extended are
# added for parity so the policy is consistent across vendors rather than an
# accident of which ones were known when the file was written.
#
# Google-Extended additionally governs whether our pages can ground AI Overviews
# and Gemini answers — Google does not separate that from training, and losing
